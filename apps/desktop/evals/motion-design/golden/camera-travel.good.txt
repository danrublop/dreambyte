export default function Scene() {
  const frame = useCurrentFrame();
  const { fps } = useVideoConfig();
  const t = frame / fps;
  const W = WIDTH, H = HEIGHT;
  const eio = (x) => x <= 0 ? 0 : x >= 1 ? 1 : (x < 0.5 ? 4*x*x*x : 1 - Math.pow(-2*x+2,3)/2);
  const cl = (x) => Math.max(0, Math.min(1, x));
  const seg = (s, e) => eio(cl((t - s) / (e - s)));

  const S1 = 760, S2 = 2280;
  const push = seg(1.9, 3.5);
  const focusX = S1 + (S2 - S1) * push;
  const scale = 1.06 - 0.06 * push;
  const world = {
    position: 'absolute', left: 0, top: 0, width: 3400, height: H,
    transformOrigin: '0 0', transform: `translate(${W/2 - focusX*scale}px, ${H/2 - (H*0.5)*scale}px) scale(${scale})`,
  };
  const s1Exit = seg(2.6, 3.7);
  const s2In = seg(2.0, 3.2);

  return (
    <AbsoluteFill style={{ background: '#000', overflow: 'hidden', fontFamily: 'Inter' }}>
      <div style={world}>
        <div style={{ position: 'absolute', left: S1 - 540, top: H*0.5 - 200, width: 1080,
          opacity: 1 - s1Exit, transform: `translateY(${-44*s1Exit}px)`, filter: `blur(${12*s1Exit}px)`,
          fontSize: 96, fontWeight: 800, color: '#f4f7fb' }}>
          Every AI video you generate
        </div>
        <div style={{ position: 'absolute', left: S2 - 540, top: H*0.5 - 120, width: 1080,
          opacity: s2In, transform: `translateY(${(1-s2In)*30}px)`,
          fontSize: 120, fontWeight: 800, color: '#fff' }}>
          a cost you don't see
        </div>
      </div>
    </AbsoluteFill>
  );
}
